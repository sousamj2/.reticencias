* Vertico and friends
** Vertico main package
#+begin_src emacs-lisp
(use-package vertico
  :ensure t
  :custom
  (vertico-cycle t)
  :init
  (vertico-mode))
#+end_src

  (use-package marginalia

** Orderless
Searching without using a specific order ffile can see also find-file
#+begin_src emacs-lisp
  (use-package orderless
    :ensure t
    :custom
    (completion-styles '(orderless basic))
    (completion-category-defaults nil)
    (completion-category-overrides
     '((file (styles basic partial-completion)))))

** Consult
#+begin_src emacs-lisp
  (use-package consult
    :ensure t
    :bind
    (("C-x b" . consult-buffer)
     ("C-s" . consult-line)
     ("C-x r b" . consult-bookmark)
     ;; ("M-g g" . consult-goto-line) Setting up in personal to [f4]
     ("M-g i" . consult-imenu)
     ("M-s r" . consult-ripgrep)))
#+end_src
** Embark
#+begin_src emacs-lisp
(use-package embark
  :ensure t
  :bind
  (("C-," . embark-act)
   ("C-;" . embark-dwim)
   ("C-h B" . embark-bindings)))
#+end_src
