** Setup functions
#+begin_src emacs-lisp
  (defun mj/org-mode-setup ()
    (message "Calling my org setup")
    (org-indent-mode)
    (variable-pitch-mode 1)
    (visual-line-mode 1))

  (defun mj/org-font-setup ()
    (dolist (face '((org-level-1 . 1.3)
  		  (org-level-2 . 1.1)
  		  (org-level-3 . 1.05)
  		  (org-level-4 . 1.0)
  		  (org-level-5 . 1.0)
  		  (org-level-6 . 1.0)
  		  (org-level-7 . 1.0)
  		  (org-level-8 . 1.0)))
      (set-face-attribute (car face) nil :font "Cantarell" :height (cdr face)))
    
    ;; Replace list hyphen with dot
    (font-lock-add-keywords 'org-mode
                            '(("^ *\\([-]\\) "
                               (0 (prog1 () (compose-region (match-beginning 1) (match-end 1) "•")))))))
#+end_src
** Start configuration
#+begin_src emacs-lisp
  (use-package org
    :pin org
    :hook (org-mode . mj/org-mode-setup))
#+end_src
