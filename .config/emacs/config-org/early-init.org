# ** Add hook for org-mode
# Execute my org-babel-tangle when the file is the a conf-org-file.
# #+begin_src emacs-lisp
# (add-hook 'org-mode-hook
#           (lambda ()
#             (when (mj/config-org-file-p)
#               (add-hook 'after-save-hook
#                         #'mj/org-babel-tangle-config
#                         nil
#                         t))))

# ;; (add-hook HOOK FUNCTION &optional DEPTH LOCAL)
# ;; HOOK is the hook name:
# ;; Whenever this happens (start org-mode, after saving the file), execute the
# ;; FUNCTION that needs to be run with a
# ;; DEPTH priority in the list of functions running the hook and only run in the
# ;; LOCAL (current) buffer if t or globally if nil.
# ;; DEPTH and LOCAL are &optional parameters in the function with the default value of nil.
# ;; In general all &optional parameters have a nil value but can be explicitly set inside the function:
# ;; (setq b (or b 42))
# ;; Sets the value to either the value provided in the argument if not nil or to 42.

# ;; In this lambda function, whatever evaluation comes after, treat it as a function.
# ;; In practice...
# ;; lambda creates function
# ;;         ↓
# ;; add-hook stores function
# ;;         ↓
# ;; event happens
# ;;         ↓
# ;; Emacs calls function
# ;;         ↓
# ;; lambda body executes
# #+end_src
