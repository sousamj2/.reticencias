** Theme at startup
*** Using time-based theme
#+begin_src emacs-lisp
  (defun mj/theme-at-startup-time ()
    "Load a the  me based on the time of day."
    (let ((hour (string-to-number (format-time-string "%H"))))
      (if (or (>= hour 19)
              (< hour 7))
          (mj/dark-theme)
        (mj/light-theme)))) 
#+end_src
*** With system-based theme
#+begin_src emacs-lisp
(defun mj/theme-at-startup-system ()
  "Load my theme acc       ording to the GNOME color scheme."
  (let ((scheme (string-trim
                 (shell-command-to-string
                  "gsettings get org.gnome.desktop.interface color-scheme"))))
    (cond
     ((string= scheme "'prefer-dark'")
      (mj/dark-theme))
     ((string= scheme "'default'")
      (mj/light-theme)))))
#+end_src

** Add-on: count number of lines
This allows to count how many lines does a function in C/C++ have.
This might be also useful for python mode. To discover later.
Placed here since the idea is to add this to the doom-modeline.
#+begin_src emacs-lisp
  (defun mj/c-function-line-count ()
    "Return the current C function's line count."
    (when (derived-mode-p 'c-mode 'c++-mode)
      (save-excursion
        (c-beginning-of-defun)
        (let ((start (point)))
          (c-end-of-defun)
          (let ((end (point)))
            (format " Func:%d/25"
                    (- (count-lines start end) 3)))))))

  (doom-modeline-def-segment mj/c-function-lines
    (mj/c-function-line-count))


  (doom-modeline-def-modeline 'mj/main
    '(eldoc bar window-state workspace-name window-number modals matches
            follow buffer-info remote-host buffer-position word-count
            parrot selection-info)
    '(compilation objed-state misc-info mj/c-function-lines
                  project-name persp-name battery grip irc mu4e gnus github
                  debug repl lsp minor-modes input-method indent-info
                  buffer-encoding major-mode process vcs check time))

  (doom-modeline-set-modeline 'mj/main)

    :commands (helpful-callable helpful-variable helpful-command helpful-key)
    :ensure t
    :custom
    (counsel-describe-function-function #'helpful-callable)
    (counsel-describe-variable-function #'helpful-variable)
    :bind
    ([remap describe-function] . counsel-describe-function)
    ([remap describe-command] . helpful-command)
