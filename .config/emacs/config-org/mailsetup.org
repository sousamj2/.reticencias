  (defun mj/mu4e-update-mail-and-index (&optional run-in-background)
    "Update mail for the current context, or all accounts if no context is active."
    (interactive "P")
    (let ((mu4e-get-mail-command
  	 (if-let ((context (mu4e-context-current)))
  	     (format "mbsync %s"
  		     (substring (mu4e-context-name context) 1)) ; (mu4e-context-name context))
  	   "mbsync -a")))
      (mu4e-update-mail-and-index run-in-background)))

  (with-eval-after-load 'mu4e
    (define-key mu4e-main-mode-map (kbd "U")
                #'mj/mu4e-update-mail-and-index))

** Private email files variable

(defvar mj/emails-file

  "Private email account configuration.")

** Function to read file
This function reads all the files in the csv file provided above.
#+begin_src emacs-lisp
(defun mj/read-emails-csv ()
  "Read email accounts from `mj/emails-file'."
  (when (file-exists-p mj/emails-file)
    (with-temp-buffer
      (insert-file-contents mj/emails-file)
      (let ((lines (split-string (buffer-string) "\n" t))) ; get each line
        (mapcar
         (lambda (line)
           (split-string line "," t)) ; get each parameter
         (cdr lines))))))
#+end_src

** Function to create account
This function will take all the information of one line and fill out all the parameters required
#+begin_src emacs-lisp
  (defun mj/mu4e-account-folder (account folder)
    "Return FOLDER prefixed with ACCOUNT's Maildir."
    (concat "/"
            account
            (if (string-prefix-p "/" folder)
                folder
              (concat "/" folder))))


  (defun mj/make-mu4e-context (account)
    "Create a mu4e context from ACCOUNT."

    (pcase-let
        ((`(,context-name
            ,_pass-dir
            ,user-email
            ,user-name
            ,smtp-server
            ,smtp-port
            ,smtp-type
            ,sent
            ,drafts
            ,bin
            ,refile)
          account))

      (let ((maildir-name (substring context-name 1)))

        (make-mu4e-context
         :name context-name

         :match-func
         (lambda (msg)
           (when msg
             (string-prefix-p
              (concat "/" maildir-name)
              (mu4e-message-field msg :maildir))))

         :vars
         `((user-mail-address . ,user-email)
           (user-full-name . ,user-name)

           (mu4e-sent-folder
            . ,(mj/mu4e-account-folder maildir-name sent))
           (mu4e-drafts-folder
            . ,(mj/mu4e-account-folder maildir-name drafts))
           (mu4e-trash-folder
            . ,(mj/mu4e-account-folder maildir-name bin))
           (mu4e-refile-folder
            . ,(mj/mu4e-account-folder maildir-name refile))

           (mu4e-sent-messages-behavior . delete)

           (smtpmail-smtp-server . ,smtp-server)
           (smtpmail-smtp-service . ,(string-to-number smtp-port))
           (smtpmail-stream-type . ,(intern smtp-type))
           (smtpmail-smtp-user . ,user-email)
           (smtpmail-auth-supported . (login)))))))
#+end_src

** Add shortcuts for using in mu4e
#+begin_src emacs-lisp
  (defun mj/update-mu4e-shortcuts ()
    "Update maildir shortcuts for the current mu4e context."
    (let ((context (mu4e-context-current)))
      (when context
  (setq mu4e-maildir-shortcuts
          `((,(concat "/"
                      (substring
                       (mu4e-context-name
                        (mu4e-context-current))
                       1)
                      "/INBOX") . ?i)
            (,mu4e-drafts-folder . ?d)
            (,mu4e-sent-folder   . ?s)
            (,mu4e-refile-folder . ?a)
            (,mu4e-trash-folder  . ?t))))))
    
  (add-hook 'mu4e-context-changed-hook
           #'mj/update-mu4e-shortcuts)

** Get the smtpmail authorization
This is just to retrieve the list of smtp servers that we are going to use to send emails.
Without duplication (multiple gmails, etc) we just replace . with \\. in the server name.
#+begin_src emacs-lisp
;  Index 4 of the csv is the smtp server name
  (defconst mj/email-smtp-server-index 4)

  (setq smtpmail-servers-requiring-authorization
      (mapconcat
       #'regexp-quote
       (delete-dups
        (mapcar
         (lambda (account)
           (nth mj/email-smtp-server-index account))
         mj/email-accounts))
       "\\|"))
#+end_src
