              '(".git"
        "__pycache__"

** Ignore extensions
#+begin_src emacs-lisp
  (setq list-of-extensions
        '(".o" ; object files
        ".a" ; static libraries
        ".so" ; dinamic libraries
        ".pyc" ; python compilation
        ".pyo" ; python objects
        ".class" ; C++ classes
        ))
#+end_src
** emacs saves
#+begin_src emacs-lisp
  (setq list-emacs-saves
     '("~$" ; emacs backup saves
        "#.*#$" ; emacs auto saves
        ))
#+end_src

** Projectile configuration
Allows to search within our projects (mine are within ~/work/ folder)
#+begin_src emacs-lisp
  (use-package projectile
    :diminish projectile-mode
    :config (projectile-mode)
    :custom
    ((projectile-completion-system 'vertico)
     (projectile-globally-ignored-directories list-of-directories)
     (projectile-globally-ignored-file-suffixes list-of-extensions)
     (projectile-global-ignore-file-patterns list-emacs-saves))
    :bind-keymap
    ("C-c p" . projectile-command-map)
    :init
    ;(when (file-directory-p "~/work/")
    (setq projectile-project-search-path '("~/work/" "~/.config/emacs/"))
;      (setq projectile-project-search-path '("~/.config/emacs/"))
    (setq projectile-switch-project-action #'projectile-dired))

#+end_src
