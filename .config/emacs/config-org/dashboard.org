* Startup
#+begin_src emacs-lisp
  (defun mj/display-startup-time ()
    (message "My livEmacs loaded in %s with %d garbage collections."
  	   (format "%.2f seconds"
  		   (float-time
  		    (time-subtract after-init-time before-init-time)))
  	   gcs-done))

  (add-hook 'emacs-startup-hook #'mj/display-startup-time)
#+end_src
