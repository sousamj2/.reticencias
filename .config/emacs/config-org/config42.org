  (defvar mj/42-loaded nil)

  (defun mj/load-42 ()
    "Load my 42 emacs extensions once."
    (unless mj/42-loaded
      (load (expand-file-name "el42/list.el" user-emacs-directory))
      (load (expand-file-name "el42/string.el" user-emacs-directory))
      (load (expand-file-name "el42/comments.el" user-emacs-directory))
      (load (expand-file-name "el42/header.el" user-emacs-directory))
      
      (global-set-key (kbd "<C-f1>") 'header-insert)
      (setq mj/42-loaded t)))

  (add-hook 'c-mode-hook #'mj/load-42)
  (add-hook 'c++-mode-hook #'mj/load-42)

* Process tabs in function and variable declaration.
#+begin_src emacs-lisp
  (defcustom mj/c-types
    '("void"
      "char"
      "int"
      "float"
      "double"
      "long")
    "C types recognized by `mj/c-align-42'."
    :type '(repeat string)
    :group 'c)

  (defun mj/c-align-42 ()
    "Align C declarations according to the 42 style."
    (interactive)
    (when (derived-mode-p 'c-mode 'c++-mode)
      (let* ((extra-types
              (when current-prefix-arg
                (let ((input
                       (read-string
                        "Additional types (space separated): ")))
                  (split-string input "[ \t,]+" t))))
             (types (append extra-types mj/c-types))
             (type-regexp (regexp-opt types 'words)))

        (save-excursion

          (if (c-defun-name)

              ;; Function definition.
              (progn
                (c-beginning-of-defun)

                ;; Return type -> TAB -> function name.
                (when (looking-at
                       (concat "^\\s-*"
                               type-regexp
                               "\\s-+"))
                  (goto-char (match-end 0))
                  (delete-horizontal-space)
                  (insert "\t"))

                ;; Find opening brace.
                (when (re-search-forward "{" nil t)
                  (forward-line 1)

                  ;; Process declarations until blank line.
                  (while (and (not (eobp))
                              (not (looking-at-p "^\\s-*$")))

                    (beginning-of-line)

                    (when (looking-at
                           (concat "^\\s-*"
                                   type-regexp
                                   "\\s-+"))

                      ;; Remember the type itself.
                      (let ((type (match-string-no-properties 0)))

                        ;; Normalize indentation.
                        (unless (looking-at "^\t")
                          (delete-horizontal-space)
                          (insert "\t"))

                        ;; Find the type again, now that indentation
                        ;; has been normalized.
                        (when (looking-at
                               (concat "^\t"
                                       type-regexp
                                       "\\s-+"))
                          (goto-char (match-end 0))
                          (delete-horizontal-space)
                          (insert "\t"))))

                    (forward-line 1))))

            ;; Function prototype.
            (beginning-of-line)

            (when (looking-at
                   (concat "^\\s-*"
                           type-regexp
                           "\\s-+"))
              (goto-char (match-end 0))
              (delete-horizontal-space)
              (insert "\t")))))))
#+end_src
